sabled = false;
    if (isSample) els.contentSelect.value = 'solo';
    updateAnalyzeButton();
    resetAnalysisUI();
    drawWaveform(buffer);
    drawPiano();
    setStatus('歌曲已就緒，開始尋找旋律。', 0);
  } catch (err) {
    console.error(err);
    alert('瀏覽器無法解碼這個音訊檔。若是 M4A，請改用 Chrome / Edge 或轉成 MP3/WAV。');
  }
}

  els.noteCount.textContent = '—';

  els.durationMetric.textContent = '—';

  els.nextPage.disabled = true;
  drawFalling(0);
}
